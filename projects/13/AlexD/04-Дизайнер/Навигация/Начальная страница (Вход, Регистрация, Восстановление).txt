@startuml
title Диаграмма состояний: Начальная страница (Вход, Регистрация, Восстановление)

skinparam state {
  BackgroundColor White
  BorderColor Black
}

[*] --> ОжиданиеВыбора

state "Ожидание выбора" as ОжиданиеВыбора {
  state "Форма входа\n(email, пароль)" as ФормаВхода
  state "Форма регистрации\n(имя, email, пароль)" as ФормаРегистрации
  state "Форма восстановления\n(ввод email)" as ФормаВосстановления
}

' Выбор действия на стартовом экране
ОжиданиеВыбора --> ФормаВхода : Вход
ОжиданиеВыбора --> ФормаРегистрации : Регистрация
ОжиданиеВыбора --> ФормаВосстановления : Восстановление пароля

' --- ВЕТКА ВХОДА ---
ФормаВхода --> ПроверкаВхода : Кнопка "Войти"
state "Проверка данных" as ПроверкаВхода
ПроверкаВхода --> ГлавнаяСтраница : [Данные верны]
ПроверкаВхода --> ОшибкаВхода : [Данные неверны]
ОшибкаВхода --> ФормаВхода : Кнопка "Назад"

' --- ВЕТКА РЕГИСТРАЦИИ ---
ФормаРегистрации --> ПроверкаРегистрации : Кнопка "Зарегистрироваться"
state "Проверка данных" as ПроверкаРегистрации
ПроверкаРегистрации --> ВводКода : [Данные корректны]
ПроверкаРегистрации --> ОшибкаРегистрации : [Данные некорректны]
ОшибкаРегистрации --> ФормаРегистрации : Кнопка "Назад"

' --- ВЕТКА ВОССТАНОВЛЕНИЯ ПАРОЛЯ ---
ФормаВосстановления --> ПроверкаEmail : Кнопка "Отправить код"
state "Проверка email" as ПроверкаEmail
ПроверкаEmail --> ОшибкаEmail : [Email не найден]
ОшибкаEmail --> ФормаВосстановления : Кнопка "Назад"
ПроверкаEmail --> ВводКода : [Email найден, код отправлен]

' --- ОБЩИЙ БЛОК ВВОДА КОДА (для регистрации и восстановления) ---
state "Ввод кода из письма" as ВводКода
ВводКода --> ПроверкаКода : Кнопка "Подтвердить"

state "Проверка кода" as ПроверкаКода
ПроверкаКода --> ОшибкаКода : [Код неверен/просрочен]
ОшибкаКода --> ВводКода : Кнопка "Назад"

ПроверкаКода --> ВводНовогоПароля : [Код верен, для восстановления]
ПроверкаКода --> ГлавнаяСтраница : [Код верен, для регистрации]

' --- ВЕТКА ВВОДА НОВОГО ПАРОЛЯ (только для восстановления) ---
state "Ввод нового пароля" as ВводНовогоПароля
ВводНовогоПароля --> ГлавнаяСтраница : Кнопка "Сохранить"

@enduml
