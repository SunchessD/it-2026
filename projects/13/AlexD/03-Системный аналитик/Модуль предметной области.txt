  +ID
  +Название
  +ЗапросТехник
}

class "Позиция" as Position {
  +ID
  +Название
  +ЗапросТехник
}

class "Тег" as Tag {

  +Название
  +ЗапросТехник
}

' ===== БЛОКИ КОЛЛЕКЦИИ (Лента) =====
abstract class "Блок коллекции" as CollBlock {
  +ID
  +Порядок (sort_order)
}

class "Карточка техники" as Technique {
  +ID
  +Категория
  +Позиция
  +Тег

  +Описание

  +Текст
}

' ===== КОЛЛЕКЦИИ =====
class "Коллекция" as Collection {
  +ID
  +Название
  +Тип (Личная / Системная)
}

' ===== КАЛЕНДАРЬ И РАСПИСАНИЕ =====
class "Календарь" as Calendar {
  +ID
  +Месяц
  +Год
  +Даты
}

class "День" as Day {
  +ID
  +Дата
  +Расписание
}

' ===== БЛОКИ ДНЯ (Расписание) =====
abstract class "Блок дня" as DayBlock {
  +ID
  +Порядок (sort_order)
}

class "Текст в расписании" as DayTextBlock {
  +Текст
}

' Календарь и дни
Calendar "1" *-- "28..31" Day 
Day "1" *-- "0..15" DayBlock

' Блоки дня
DayBlock "1" <|-- "1" DayTextBlock

' Коллекция напрямую связана с Блоком дня (как один блок)
Collection "1" -- "1" DayBlock
