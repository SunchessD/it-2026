# Команды для интерактивного redis-cli. Ключ относится только к демонстрации.
# При Docker Compose: docker compose exec redis redis-cli
SET demo:yudin:summary "720.00" EX 60
GET demo:yudin:summary
TTL demo:yudin:summary
DEL demo:yudin:summary
GET demo:yudin:summary
TTL demo:yudin:summary
# После DEL GET возвращает nil, а TTL возвращает -2.
